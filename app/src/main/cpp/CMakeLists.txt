cmake_minimum_required(VERSION 3.22.1)
project(chipplayer LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GME_BUILD_SHARED ON CACHE BOOL "" FORCE)
set(GME_BUILD_STATIC OFF CACHE BOOL "" FORCE)
set(GME_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(GME_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(VENDOR ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party)
add_subdirectory(${VENDOR}/game-music-emu gme)
set(unarr_VERSION_MAJOR 1)
set(unarr_VERSION_MINOR 2)
set(unarr_VERSION_PATCH 0)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/unarr-include)
configure_file(${VENDOR}/unarr/unarr.h.in ${CMAKE_CURRENT_BINARY_DIR}/unarr-include/unarr.h @ONLY)
add_library(unarr SHARED
    ${VENDOR}/unarr/common/conv.c ${VENDOR}/unarr/common/crc32.c ${VENDOR}/unarr/common/stream.c ${VENDOR}/unarr/common/unarr.c
    ${VENDOR}/unarr/rar/rar.c ${VENDOR}/unarr/rar/parse-rar.c ${VENDOR}/unarr/rar/uncompress-rar.c ${VENDOR}/unarr/rar/filter-rar.c ${VENDOR}/unarr/rar/huffman-rar.c ${VENDOR}/unarr/rar/rarvm.c
    ${VENDOR}/unarr/lzmasdk/Ppmd7.c ${VENDOR}/unarr/lzmasdk/Ppmd7Dec.c ${VENDOR}/unarr/lzmasdk/Ppmd7aDec.c)
target_include_directories(unarr PUBLIC ${CMAKE_CURRENT_BINARY_DIR}/unarr-include)
target_compile_definitions(unarr PRIVATE UNARR_EXPORT_SYMBOLS PUBLIC UNARR_IS_SHARED_LIBRARY)
add_library(player SHARED player.cpp rar_reader.cpp psf_decoder.cpp midi_decoder.cpp
    ${VENDOR}/psf-core/psx.cc ${VENDOR}/psf-core/psx_hw.cc ${VENDOR}/psf-core/peops/spu.cc)
target_compile_features(player PRIVATE cxx_std_11)
target_include_directories(player PRIVATE ${VENDOR}/psf-core ${VENDOR}/tinysoundfont)
target_compile_options(player PRIVATE -fno-strict-aliasing -fwrapv)
target_link_libraries(player PRIVATE gme::gme unarr z)
if(NOT ANDROID)
    target_include_directories(player PRIVATE $ENV{JAVA_HOME}/include $ENV{JAVA_HOME}/include/linux)
    target_link_libraries(player PRIVATE pthread)
endif()
