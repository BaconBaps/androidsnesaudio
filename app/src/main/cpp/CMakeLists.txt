cmake_minimum_required(VERSION 3.22.1)
project(chipamp9x LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GME_BUILD_SHARED ON CACHE BOOL "" FORCE)
set(GME_BUILD_STATIC OFF CACHE BOOL "" FORCE)
set(GME_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(GME_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(VENDOR ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party)
add_subdirectory(${VENDOR}/game-music-emu gme)
# Portable libvgm player with a broad sound-device collection; no audio backends.
set(BUILD_LIBAUDIO OFF CACHE BOOL "" FORCE)
set(BUILD_PLAYER OFF CACHE BOOL "" FORCE)
set(BUILD_VGM2WAV OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(USE_SANITIZERS OFF CACHE BOOL "" FORCE)
# Enable the broad device collection except OPL4's GPL-2.0-only core, which
# cannot be combined with this APK's GPLv3 components.
set(SNDEMU__ALL OFF CACHE BOOL "" FORCE)
file(STRINGS ${VENDOR}/libvgm/emu/CMakeLists.txt VGM_DEVICE_OPTIONS REGEX "^option\\(SNDEMU_.*_ALL ")
foreach(line IN LISTS VGM_DEVICE_OPTIONS)
    string(REGEX REPLACE "^option\\(([^ ]+).*" "\\1" device "${line}")
    if(NOT device STREQUAL "SNDEMU_YMF278B_ALL" AND NOT device STREQUAL "SNDEMU__ALL")
        set(${device} ON CACHE BOOL "" FORCE)
    endif()
endforeach()
set(SNDEMU_YMF278B_ALL OFF CACHE BOOL "" FORCE)
set(UTIL_THREADING OFF CACHE BOOL "" FORCE)
set(UTIL_CHARSET_CONV OFF CACHE BOOL "" FORCE)
set(LIBRARY_TYPE STATIC CACHE STRING "" FORCE)
add_subdirectory(${VENDOR}/libvgm libvgm)
# libvgm changes output directories globally. Keep JNI libraries in their usual locations.
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "")
set(unarr_VERSION_MAJOR 1)
set(unarr_VERSION_MINOR 2)
set(unarr_VERSION_PATCH 0)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/unarr-include)
configure_file(${VENDOR}/unarr/unarr.h.in ${CMAKE_CURRENT_BINARY_DIR}/unarr-include/unarr.h @ONLY)
add_library(unarr SHARED
    ${VENDOR}/unarr/common/conv.c ${VENDOR}/unarr/common/crc32.c ${VENDOR}/unarr/common/stream.c ${VENDOR}/unarr/common/unarr.c
    ${VENDOR}/unarr/rar/rar.c ${VENDOR}/unarr/rar/parse-rar.c ${VENDOR}/unarr/rar/uncompress-rar.c ${VENDOR}/unarr/rar/filter-rar.c ${VENDOR}/unarr/rar/huffman-rar.c ${VENDOR}/unarr/rar/rarvm.c
    ${VENDOR}/unarr/lzmasdk/Ppmd7.c ${VENDOR}/unarr/lzmasdk/Ppmd7Dec.c ${VENDOR}/unarr/lzmasdk/Ppmd7aDec.c)
target_include_directories(unarr PUBLIC ${CMAKE_CURRENT_BINARY_DIR}/unarr-include)
target_compile_definitions(unarr PRIVATE UNARR_EXPORT_SYMBOLS PUBLIC UNARR_IS_SHARED_LIBRARY)
# Portable cached interpreter on both ARM64 and x86_64; no executable-memory JIT.
file(GLOB_RECURSE USF_SOURCES CONFIGURE_DEPENDS ${VENDOR}/lazyusf2/*.c)
list(FILTER USF_SOURCES EXCLUDE REGEX "/(bench|instr_counters)\\.c$")
add_library(lazyusf2 STATIC ${USF_SOURCES})
target_include_directories(lazyusf2 PUBLIC ${VENDOR}/lazyusf2)
# Sideload APKs use the debug build type: emulation must still run optimised.
# Keep ordinary optimisation, not fast-math, to retain integer/audio behaviour.
target_compile_options(lazyusf2 PRIVATE -O2 -fno-strict-aliasing -fwrapv)
target_link_libraries(lazyusf2 PUBLIC m)
add_library(player SHARED player.cpp vgm_decoder.cpp vgm_text.cpp rar_reader.cpp psf_decoder.cpp midi_decoder.cpp usf_decoder.cpp psx_audio.cpp
    ${VENDOR}/psf-core/psx.cc ${VENDOR}/psf-core/psx_hw.cc ${VENDOR}/psf-core/peops/spu.cc)
set_source_files_properties(psx_audio.cpp vgm_decoder.cpp PROPERTIES COMPILE_OPTIONS "-O2")
target_compile_features(player PRIVATE cxx_std_11)
target_include_directories(player PRIVATE ${VENDOR}/psf-core ${VENDOR}/tinysoundfont)
target_compile_options(player PRIVATE -fno-strict-aliasing -fwrapv)
target_link_libraries(player PRIVATE gme::gme unarr z lazyusf2 vgm-player)
if(NOT ANDROID)
    add_executable(xsf_timing_test ${VENDOR}/../tests/XsfTimingTest.cpp)
    target_include_directories(xsf_timing_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_features(xsf_timing_test PRIVATE cxx_std_11)
    target_include_directories(player PRIVATE $ENV{JAVA_HOME}/include $ENV{JAVA_HOME}/include/linux)
    target_link_libraries(player PRIVATE pthread)
endif()

target_compile_options(vgm-emu PRIVATE -O2 -fno-strict-aliasing -fwrapv)
target_compile_options(vgm-player PRIVATE -O2 -fno-strict-aliasing -fwrapv)
